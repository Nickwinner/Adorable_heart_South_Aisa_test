on_actions = {
	on_daily_NRR = {
		effect = {
			if = {
				limit = {
					has_country_flag = NRR_has_active_bill
				}
				subtract_from_variable = { NRR_bill_days_left = 1 }
				if = {
					limit = {
						check_variable = { NRR_bill_days_left < 1 }
					}
					NRR_call_vote_effect = yes
				}
			}
		}
	}
}
