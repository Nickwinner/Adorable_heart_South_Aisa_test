NRR_init_parliament_if_needed = {
	if = {
		limit = {
			NOT = { has_country_flag = NRR_parliament_initialized }
		}
		set_country_flag = NRR_parliament_initialized

		# Default House Seats (Total = 350)
		set_variable = { NRR_seats_dem = 175 }
		set_variable = { NRR_seats_soc = 95 }
		set_variable = { NRR_seats_peo = 80 }
		set_variable = { NRR_seats_rec = 0 }

		# Default Senate Seats (Total = 50)
		set_variable = { NRR_sen_seats_dem = 21 }
		set_variable = { NRR_sen_seats_peo = 18 }
		set_variable = { NRR_sen_seats_soc = 11 }
		set_variable = { NRR_sen_seats_rec = 0 }

		# Default: No Active Bill ("暂无正在审议的法案")
		clr_country_flag = NRR_has_active_bill
		set_variable = { NRR_active_bill_id = 0 }
		set_variable = { NRR_bill_days_left = 0 }

		# Initial Supporting Members = 0 (House & Senate)
		set_variable = { NRR_bill_yes_dem = 0 }
		set_variable = { NRR_bill_yes_soc = 0 }
		set_variable = { NRR_bill_yes_peo = 0 }
		set_variable = { NRR_bill_yes_rec = 0 }
		set_variable = { NRR_sen_bill_yes_dem = 0 }
		set_variable = { NRR_sen_bill_yes_soc = 0 }
		set_variable = { NRR_sen_bill_yes_peo = 0 }
		set_variable = { NRR_sen_bill_yes_rec = 0 }
	}
}

NRR_calculate_parliament_all = {
	NRR_calculate_house_dots = yes
	NRR_calculate_senate_dots = yes
	NRR_calculate_bill_votes = yes
}

NRR_calculate_house_dots = {
	clear_array = NRR_house_dots

	# 1. Reconstruction Committee (Left / Frame 4)
	set_temp_variable = { t_idx = 0 }
	while_loop_effect = {
		limit = { check_variable = { t_idx < NRR_seats_rec } }
		add_to_array = { array = NRR_house_dots value = 4 }
		add_to_temp_variable = { t_idx = 1 }
	}

	# 2. Socialist Party (Centre-Left / Frame 2)
	set_temp_variable = { t_idx = 0 }
	while_loop_effect = {
		limit = { check_variable = { t_idx < NRR_seats_soc } }
		add_to_array = { array = NRR_house_dots value = 2 }
		add_to_temp_variable = { t_idx = 1 }
	}

	# 3. Democratic Republic Party (Centre-Right / Frame 1)
	set_temp_variable = { t_idx = 0 }
	while_loop_effect = {
		limit = { check_variable = { t_idx < NRR_seats_dem } }
		add_to_array = { array = NRR_house_dots value = 1 }
		add_to_temp_variable = { t_idx = 1 }
	}

	# 4. People's Party (Right / Frame 3 - Grey)
	set_temp_variable = { t_idx = 0 }
	while_loop_effect = {
		limit = { check_variable = { t_idx < NRR_seats_peo } }
		add_to_array = { array = NRR_house_dots value = 3 }
		add_to_temp_variable = { t_idx = 1 }
	}

	# 5. Fill remainder up to 350 (Frame 5 - Dark slate/empty)
	while_loop_effect = {
		limit = { check_variable = { NRR_house_dots^num < 350 } }
		add_to_array = { array = NRR_house_dots value = 5 }
	}
}

NRR_calculate_bill_votes = {
	# Clamp supporting members to each party's current House seats
	clamp_variable = { var = NRR_bill_yes_dem min = 0 max = NRR_seats_dem }
	clamp_variable = { var = NRR_bill_yes_soc min = 0 max = NRR_seats_soc }
	clamp_variable = { var = NRR_bill_yes_peo min = 0 max = NRR_seats_peo }
	clamp_variable = { var = NRR_bill_yes_rec min = 0 max = NRR_seats_rec }

	# House Projected Yes Votes = sum of supporting members across parties
	set_temp_variable = { temp_house_yes = NRR_bill_yes_dem }
	add_to_temp_variable = { temp_house_yes = NRR_bill_yes_soc }
	add_to_temp_variable = { temp_house_yes = NRR_bill_yes_peo }
	add_to_temp_variable = { temp_house_yes = NRR_bill_yes_rec }
	set_variable = { NRR_house_yes_votes = temp_house_yes }

	# Progress Bar Percentage (0 to 100 for HOI4 progressbartype frame)
	set_variable = { NRR_vote_progress_pct = NRR_house_yes_votes }
	multiply_variable = { NRR_vote_progress_pct = 100 }
	divide_variable = { NRR_vote_progress_pct = 350 }
	round_variable = NRR_vote_progress_pct
	clamp_variable = { var = NRR_vote_progress_pct min = 0 max = 100 }

	# Clamp supporting members to each party's current Senate seats
	clamp_variable = { var = NRR_sen_bill_yes_dem min = 0 max = NRR_sen_seats_dem }
	clamp_variable = { var = NRR_sen_bill_yes_soc min = 0 max = NRR_sen_seats_soc }
	clamp_variable = { var = NRR_sen_bill_yes_peo min = 0 max = NRR_sen_seats_peo }
	clamp_variable = { var = NRR_sen_bill_yes_rec min = 0 max = NRR_sen_seats_rec }

	# Senate Projected Yes Votes = sum of Senate supporting members across parties
	set_temp_variable = { temp_sen_yes = NRR_sen_bill_yes_dem }
	add_to_temp_variable = { temp_sen_yes = NRR_sen_bill_yes_soc }
	add_to_temp_variable = { temp_sen_yes = NRR_sen_bill_yes_peo }
	add_to_temp_variable = { temp_sen_yes = NRR_sen_bill_yes_rec }
	set_variable = { NRR_senate_yes_votes = temp_sen_yes }
}

# Helper: Start bill countdown (if NRR_bill_days_left wasn't set by the bill, default to 42 days)
NRR_start_bill_vote = {
	set_country_flag = NRR_has_active_bill
	if = {
		limit = {
			NOT = { check_variable = { NRR_bill_days_left > 0 } }
		}
		set_variable = { NRR_bill_days_left = 42 }
	}
	NRR_calculate_bill_votes = yes
}

# Helper: Clear active bill and return to "暂无正在审议的法案"
NRR_clear_active_bill = {
	clr_country_flag = NRR_has_active_bill
	set_variable = { NRR_active_bill_id = 0 }
	set_variable = { NRR_bill_days_left = 0 }
	set_variable = { NRR_bill_yes_dem = 0 }
	set_variable = { NRR_bill_yes_soc = 0 }
	set_variable = { NRR_bill_yes_peo = 0 }
	set_variable = { NRR_bill_yes_rec = 0 }
	set_variable = { NRR_sen_bill_yes_dem = 0 }
	set_variable = { NRR_sen_bill_yes_soc = 0 }
	set_variable = { NRR_sen_bill_yes_peo = 0 }
	set_variable = { NRR_sen_bill_yes_rec = 0 }
	NRR_calculate_bill_votes = yes
}

# Voting Resolution Effect (Triggered automatically when 42-day countdown reaches 0)
NRR_call_vote_effect = {
	custom_effect_tooltip = NRR_call_vote_resolution_tt

	if = {
		limit = {
			has_country_flag = NRR_has_senate
		}
		# Bicameral System: Must pass BOTH House (>= 176) and Senate (>= 26)
		if = {
			limit = {
				check_variable = { NRR_house_yes_votes > 175 }
				check_variable = { NRR_senate_yes_votes > 25 }
			}
			NRR_bill_pass_success_effect = yes
		}
		else = {
			NRR_bill_pass_fail_effect = yes
		}
	}
	else = {
		# Unicameral System: Only needs House (>= 176)
		if = {
			limit = {
				check_variable = { NRR_house_yes_votes > 175 }
			}
			NRR_bill_pass_success_effect = yes
		}
		else = {
			NRR_bill_pass_fail_effect = yes
		}
	}

	# Return to idle state ("暂无正在审议的法案") regardless of pass or fail
	NRR_clear_active_bill = yes
}

NRR_bill_pass_success_effect = {
	add_stability = 0.05
	add_political_power = 60
	NRR_apply_bill_pass_effects = yes
}

NRR_bill_pass_fail_effect = {
	add_stability = -0.05
	add_political_power = -30
	NRR_apply_bill_fail_effects = yes
}

#  【法案专属效果区】—— 每次新增法案只需修改下面三个部分
# ========================================================

# 1) 法案通过后的专属效果（42天倒计时结束且赞成票达标时自动触发）
NRR_apply_bill_pass_effects = {
	# --- 1号法案通过效果 ---
	if = {
		limit = { check_variable = { NRR_active_bill_id = 1 } }
		set_country_flag = NRR_bill_1_passed # 记录“1号法案已通过”标记，方便国策/事件检查
		# 你也可以在这里继续加效果，比如：
		# add_ideas = 你的国家精神ID
		# country_event = { id = 你的事件ID }
	}

	# --- 如果以后有2号法案，取消下面注释并照着改即可 ---
	# if = {
	# 	limit = { check_variable = { NRR_active_bill_id = 2 } }
	# 	set_country_flag = NRR_bill_2_passed
	# }
}

# 2) 法案被否决后的专属效果（42天倒计时结束但赞成票未达标时自动触发）
NRR_apply_bill_fail_effects = {
	# --- 1号法案否决效果 ---
	if = {
		limit = { check_variable = { NRR_active_bill_id = 1 } }
		set_country_flag = NRR_bill_1_failed # 记录“1号法案被否决”标记（可选）
		# 你也可以在这里加惩罚或后续事件，比如：
		# country_event = { id = 你的否决后事件ID }
	}
}

# 3) 【提交法案入口】—— 在国策 completion_reward 或事件 option 里直接写：
#    NRR_submit_bill_1 = yes
NRR_submit_bill_1 = {
	NRR_init_parliament_if_needed = yes   # 确保议会变量已初始化（防止玩家还没点开过议会界面）
	set_variable = { NRR_active_bill_id = 1 } # 告诉议会当前审议的是“1号法案”（对应 sloc 里的 NRR_BILL_1_TITLE / DESC）

	# 设定【众议院】各党派在议案刚提出时的初始支持人数（你可以按剧情自由调整数字）
	set_variable = { NRR_bill_yes_dem = 100 } # 众议院民共党初始支持人数（当前上限175）
	set_variable = { NRR_bill_yes_soc = 50 }  # 众议院社会党初始支持人数（当前上限95）
	set_variable = { NRR_bill_yes_peo = 10 }  # 众议院人民党初始支持人数（当前上限80）
	set_variable = { NRR_bill_yes_rec = 0 }   # 众议院国家重建委员会初始支持人数（没有委员会时保持0即可）

	# 设定【参议院】各党派在议案刚提出时的初始支持人数（若实行两院制时生效）
	set_variable = { NRR_sen_bill_yes_dem = 12 } # 参议院民共党初始支持人数（初始上限21）
	set_variable = { NRR_sen_bill_yes_soc = 6 }  # 参议院社会党初始支持人数（初始上限11）
	set_variable = { NRR_sen_bill_yes_peo = 3 }  # 参议院人民党初始支持人数（初始上限18）
	set_variable = { NRR_sen_bill_yes_rec = 0 }  # 参议院国家重建委员会初始支持人数（没有委员会时保持0即可）

	# 独立设定本法案的审议倒计时天数（比如改成 30、42、60 等）
	set_variable = { NRR_bill_days_left = 42 }

	NRR_start_bill_vote = yes             # 激活议案标记 + 立即刷新赞成票数与进度条
}
