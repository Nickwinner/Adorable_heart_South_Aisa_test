LOS_init_culinary_if_needed = {
	if = {
		limit = {
			NOT = { has_country_flag = LOS_culinary_initialized }
		}
		set_country_flag = LOS_culinary_initialized
		set_variable = { LOS_cul_selected_ing = 1 }
		set_variable = { LOS_cul_selected_cook = 1 }
		set_variable = { LOS_cul_selected_chef = 1 }
		set_variable = { LOS_cul_days_left = 0 }
		set_variable = { LOS_cul_progress = 0 }
		set_variable = { LOS_cul_cd_days_left = 0 }
		set_variable = { LOS_cul_ing_cd_1 = 0 }
		set_variable = { LOS_cul_ing_cd_2 = 0 }
		set_variable = { LOS_cul_ing_cd_3 = 0 }
		set_variable = { LOS_cul_ing_cd_4 = 0 }
		set_variable = { LOS_cul_ing_cd_5 = 0 }
		set_variable = { LOS_cul_ing_cd_6 = 0 }
		set_variable = { LOS_cul_ing_cd_7 = 0 }
		set_variable = { LOS_cul_ing_cd_8 = 0 }
		set_variable = { LOS_cul_ing_cd_9 = 0 }
		set_variable = { LOS_cul_ing_cd_10 = 0 }
		set_variable = { LOS_cul_ing_cd_11 = 0 }
		set_variable = { LOS_cul_ing_cd_12 = 0 }

		# 初始已发现的 6 种迷宫食材（7~12 号初始锁定，通过对应 country_flag 自动解锁）
		clear_array = LOS_cul_unlocked_ingredients
		add_to_array = { LOS_cul_unlocked_ingredients = 1 } # 1. 大蝎子
		add_to_array = { LOS_cul_unlocked_ingredients = 2 } # 2. 走路菇
		add_to_array = { LOS_cul_unlocked_ingredients = 3 } # 3. 宝箱怪
		add_to_array = { LOS_cul_unlocked_ingredients = 4 } # 4. 曼德拉草
		add_to_array = { LOS_cul_unlocked_ingredients = 5 } # 5. 炎龙肉
		add_to_array = { LOS_cul_unlocked_ingredients = 6 } # 6. 史莱姆
	}
}

# 根据剧情 country_flag 自动解锁后续食材（7~12 号）
LOS_cul_sync_unlocked_ingredients = {
	LOS_init_culinary_if_needed = yes
	# 7. 黄泉丑鬼肉（魔都精兵）
	if = {
		limit = {
			has_country_flag = LOS_cul_unlock_shuuki
			NOT = { is_in_array = { array = LOS_cul_unlocked_ingredients value = 7 } }
		}
		add_to_array = { LOS_cul_unlocked_ingredients = 7 }

#  - t_synergy =  1 （绝配）：大成功概率 +20%，黑暗料理概率 -15%（最低0%）
#  - t_synergy =  0 （常规）：完全使用主厨基础概率（新食材不写本表也默认走 0，不会报错）
#  - t_synergy = -1 （踩雷）：黑暗料理概率 +20%，普通成功概率 -20%

LOS_cul_eval_synergy = {
	set_temp_variable = { t_synergy = 0 }

		else_if = { limit = { check_variable = { LOS_cul_active_cook = 5 } } set_temp_variable = { t_synergy = -1 } }

	else_if = {
		limit = { check_variable = { LOS_cul_active_ing = 2 } }
		if = {
			limit = {
				OR = {
					check_variable = { LOS_cul_active_cook = 2 }

				}
			}
			set_temp_variable = { t_synergy = 1 }

LOS_cul_daily_tick = {
	if = {
		limit = { has_country_flag = LOS_culinary_unlocked }
		LOS_cul_sync_unlocked_ingredients = yes
		LOS_cul_tick_cooldowns = yes
	}
	if = {
		limit = { has_country_flag = LOS_cul_research_active }
		subtract_from_variable = { LOS_cul_days_left = 1 }
		set_temp_variable = { t_done = 15 }
		subtract_from_temp_variable = { t_done = LOS_cul_days_left }
		multiply_temp_variable = { t_done = 100 }
		divide_temp_variable = { t_done = 15 }
		round_temp_variable = t_done
		set_variable = { LOS_cul_progress = t_done }
		clamp_variable = { var = LOS_cul_progress min = 0 max = 100 }

		if = {
			limit = { check_variable = { LOS_cul_days_left < 1 } }
			clr_country_flag = LOS_cul_research_active
			set_variable = { LOS_cul_days_left = 0 }
			set_variable = { LOS_cul_progress = 0 }

			# 1. 计算主厨基础概率（大成功 t_p1 / 普通成功 t_p2 / 黑暗料理 t_p3）
			if = {
				limit = { check_variable = { LOS_cul_active_chef = 1 } }
				set_temp_variable = { t_p1 = 30 }
				set_temp_variable = { t_p2 = 65 }
				set_temp_variable = { t_p3 = 5 }
			}
			else_if = {
				limit = { check_variable = { LOS_cul_active_chef = 2 } }
				set_temp_variable = { t_p1 = 45 }
				set_temp_variable = { t_p2 = 30 }
				set_temp_variable = { t_p3 = 25 }
			}
			else_if = {
				limit = { check_variable = { LOS_cul_active_chef = 3 } }
				set_temp_variable = { t_p1 = 35 }
				set_temp_variable = { t_p2 = 50 }
				set_temp_variable = { t_p3 = 15 }
			}
			else = {
				set_temp_variable = { t_p1 = 25 }
				set_temp_variable = { t_p2 = 65 }
				set_temp_variable = { t_p3 = 10 }
			}

			# 2. 叠加食材×做法契合度修正
			LOS_cul_eval_synergy = yes
			if = {
				limit = { check_variable = { t_synergy = 1 } }
				add_to_temp_variable = { t_p1 = 20 }
				subtract_from_temp_variable = { t_p3 = 15 }
				clamp_temp_variable = { var = t_p3 min = 0 max = 100 }
			}
			else_if = {
				limit = { check_variable = { t_synergy = -1 } }
				add_to_temp_variable = { t_p3 = 20 }
				subtract_from_temp_variable = { t_p2 = 20 }
				clamp_temp_variable = { var = t_p2 min = 5 max = 100 }
			}

			# 3. 无论成败，本次消耗的食材进入 120 天补货冷却
			LOS_cul_set_active_ing_cd = yes

			# 4. 按最终权重触发结果事件（成功/大成功进入 60 天灶台冷却；黑暗料理无灶台冷却）
			random_list = {
				0 = {
					modifier = { add = t_p1 }
					set_variable = { LOS_cul_cd_days_left = 60 }
					country_event = { id = LOS_culinary.1 }
				}
				0 = {
					modifier = { add = t_p2 }
					set_variable = { LOS_cul_cd_days_left = 60 }
					country_event = { id = LOS_culinary.2 }
				}
				0 = {
					modifier = { add = t_p3 }
					country_event = { id = LOS_culinary.3 }
				}
			}
		}
	}
}
